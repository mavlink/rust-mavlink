1//! Compile-time dialect adapter.
2
3use core::{convert::Infallible, marker::PhantomData};
4
5use crate::{Dialect, MavlinkVersion, Message, error::ParserError};
6
7/// Adapts a generated [`Message`] type to the instance-backed [`Dialect`] API.
8///
9/// This hidden bridge preserves the existing compile-time API while letting
10/// static and runtime dialects share reader framing and decoding logic.
11#[doc(hidden)]
12// A function-pointer marker keeps this zero-sized adapter from imposing an
13// unnecessary `'static` requirement on the generated message type.
14pub struct StaticDialect<M>(PhantomData<fn() -> M>);
15
16impl<M> StaticDialect<M> {
17    pub(crate) const fn new() -> Self {
18        Self(PhantomData)
19    }
20}
21
22impl<M: Message> Dialect for StaticDialect<M> {
23    type Message = M;
24    type EncodeError = Infallible;
25
26    fn message_id(&self, message: &Self::Message) -> u32 {
27        message.message_id()
28    }
29
30    fn extra_crc(&self, message_id: u32) -> Option<u8> {
31        Some(M::extra_crc(message_id))
32    }
33
34    fn decode(
35        &self,
36        version: MavlinkVersion,
37        message_id: u32,
38        payload: &[u8],
39    ) -> Result<Self::Message, ParserError> {
40        M::parse(version, message_id, payload)
41    }
42
43    fn encode(
44        &self,
45        version: MavlinkVersion,
46        message: &Self::Message,
47        payload: &mut [u8],
48    ) -> Result<usize, Self::EncodeError> {
49        Ok(message.ser(version, payload))
50    }
51}
